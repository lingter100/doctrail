cmake_minimum_required(VERSION 3.31)
# Prevent archive builds from discovering the containing repository or remote tools.
set(CMAKE_DISABLE_FIND_PACKAGE_Git TRUE CACHE BOOL "" FORCE)
set(GIT_EXECUTABLE "" CACHE FILEPATH "" FORCE)
set(GIT_EXE "" CACHE FILEPATH "" FORCE)
set(BUILD_COMMIT "d81235049384534c167caea52b85a694f6103d14" CACHE STRING "" FORCE)
set(BUILD_NUMBER 11429 CACHE STRING "" FORCE)
set(LLAMA_BUILD_COMMIT "d81235049384534c167caea52b85a694f6103d14" CACHE STRING "" FORCE)
set(LLAMA_BUILD_NUMBER 11429 CACHE STRING "" FORCE)
set(GGML_BUILD_COMMIT "d81235049384534c167caea52b85a694f6103d14" CACHE STRING "" FORCE)
project(summary_grammar_engine LANGUAGES C CXX)
if(NOT IS_ABSOLUTE "${LLAMA_SOURCE}" OR NOT EXISTS "${LLAMA_SOURCE}/src/llama-grammar.h")
  message(FATAL_ERROR "Provide the separately hash-verified pinned source directory")
endif()
foreach(flag BUILD_SHARED_LIBS LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS
    LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_BUILD_UI
    LLAMA_USE_PREBUILT_UI LLAMA_TOOLS_INSTALL LLAMA_TESTS_INSTALL LLAMA_OPENSSL
    LLAMA_SUBPROCESS LLAMA_LLGUIDANCE LLAMA_USE_SYSTEM_GGML
    GGML_NATIVE GGML_CCACHE GGML_OPENMP GGML_BACKEND_DL GGML_CUDA GGML_CUDA_GRAPHS
    GGML_HIP GGML_HIPBLAS GGML_METAL GGML_VULKAN GGML_SYCL GGML_OPENCL GGML_BLAS
    GGML_RPC GGML_CANN GGML_MUSA GGML_WEBGPU GGML_HEXAGON GGML_ZDNN GGML_VIRTGPU
    GGML_CPU_KLEIDIAI GGML_CPU_HBM GGML_LLAMAFILE GGML_ACCELERATE GGML_OPENMP_FETCH
    GGML_OPENVINO GGML_ET GGML_ET_SYSEMU GGML_ZENDNN GGML_VIRTGPU_BACKEND GGML_LTO
    GGML_CPU_ALL_VARIANTS GGML_CPU_REPACK GGML_BUILD_TESTS GGML_BUILD_EXAMPLES)
  set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
set(GGML_CPU ON CACHE BOOL "" FORCE)
set(GGML_STATIC ON CACHE BOOL "" FORCE)
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "" FORCE)
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)
add_subdirectory("${LLAMA_SOURCE}" pinned-llama EXCLUDE_FROM_ALL)
add_executable(summary-grammar-validator harness.cpp)
target_compile_features(summary-grammar-validator PRIVATE cxx_std_17)
target_include_directories(summary-grammar-validator PRIVATE "${LLAMA_SOURCE}/src" "${LLAMA_SOURCE}/include")
target_link_libraries(summary-grammar-validator PRIVATE llama)
if(MINGW)
  target_link_options(summary-grammar-validator PRIVATE -static)
endif()
set_target_properties(summary-grammar-validator PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
